SET SERVEROUTPUT ON;

-- ======================================================
-- EXAMPLE 1: NO_DATA_FOUND
-- ======================================================

DECLARE
    v_name customer.first_name%TYPE;
BEGIN
    SELECT first_name
    INTO v_name
    FROM customer
    WHERE customer_id = 9999;

    DBMS_OUTPUT.PUT_LINE('Customer: ' || v_name);

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('Error: Customer not found.');
END;
/

-- ======================================================
-- EXAMPLE 2: DUP_VAL_ON_INDEX
-- ======================================================

BEGIN
    INSERT INTO customer
    VALUES (
        seq_customer.NEXTVAL,
        'John',
        'Duplicate',
        '0788000001',
        'duplicate@example.com',
        'Kigali'
    );

EXCEPTION
    WHEN DUP_VAL_ON_INDEX THEN
        DBMS_OUTPUT.PUT_LINE('Error: Phone number or email already exists.');
END;
/

-- ======================================================
-- EXAMPLE 3: TOO_MANY_ROWS
-- ======================================================

DECLARE
    v_status service_order.status%TYPE;
BEGIN
    SELECT status
    INTO v_status
    FROM service_order;

    DBMS_OUTPUT.PUT_LINE(v_status);

EXCEPTION
    WHEN TOO_MANY_ROWS THEN
        DBMS_OUTPUT.PUT_LINE('Error: More than one row was returned.');
END;
/

-- ======================================================
-- EXAMPLE 4: ZERO_DIVIDE
-- ======================================================

DECLARE
    v_result NUMBER;
BEGIN
    v_result := 100 / 0;

    DBMS_OUTPUT.PUT_LINE(v_result);

EXCEPTION
    WHEN ZERO_DIVIDE THEN
        DBMS_OUTPUT.PUT_LINE('Error: Division by zero is not allowed.');
END;
/

-- ======================================================
-- EXAMPLE 5: USER-DEFINED EXCEPTION
-- ======================================================

DECLARE
    invalid_amount EXCEPTION;
    v_amount NUMBER := -50;
BEGIN

    IF v_amount < 0 THEN
        RAISE invalid_amount;
    END IF;

    DBMS_OUTPUT.PUT_LINE('Payment Amount: ' || v_amount);

EXCEPTION
    WHEN invalid_amount THEN
        DBMS_OUTPUT.PUT_LINE('Error: Payment amount cannot be negative.');
END;
/

-- ======================================================
-- EXAMPLE 6: OTHERS EXCEPTION
-- ======================================================

BEGIN
    UPDATE service
    SET price = price / 0
    WHERE service_id = 1;

EXCEPTION
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('Unexpected Error: ' || SQLERRM);
END;
/

-- ======================================================
-- EXAMPLE 7: EXCEPTION HANDLING IN A PROCEDURE
-- ======================================================

CREATE OR REPLACE PROCEDURE find_customer (
    p_customer_id IN NUMBER
)
AS
    v_name customer.first_name%TYPE;
BEGIN
    SELECT first_name
    INTO v_name
    FROM customer
    WHERE customer_id = p_customer_id;

    DBMS_OUTPUT.PUT_LINE('Customer Name: ' || v_name);

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('Customer ID does not exist.');

    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE(SQLERRM);
END;
/

-- TEST PROCEDURE
