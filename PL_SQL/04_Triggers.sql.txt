-- ======================================================
-- TRIGGER 1: CUSTOMER AUDIT
-- ======================================================

CREATE OR REPLACE TRIGGER trg_customer_audit
AFTER INSERT OR UPDATE OR DELETE
ON customer
FOR EACH ROW
BEGIN

    IF INSERTING THEN
        INSERT INTO audit_log(
            audit_id,
            table_name,
            operation,
            old_value,
            new_value,
            changed_by,
            change_date
        )
        VALUES(
            seq_audit_log.NEXTVAL,
            'CUSTOMER',
            'INSERT',
            NULL,
            'Customer ID: ' || :NEW.customer_id,
            USER,
            SYSDATE
        );

    ELSIF UPDATING THEN
        INSERT INTO audit_log(
            audit_id,
            table_name,
            operation,
            old_value,
            new_value,
            changed_by,
            change_date
        )
        VALUES(
            seq_audit_log.NEXTVAL,
            'CUSTOMER',
            'UPDATE',
            'Customer ID: ' || :OLD.customer_id,
            'Customer ID: ' || :NEW.customer_id,
            USER,
            SYSDATE
        );

    ELSIF DELETING THEN
        INSERT INTO audit_log(
            audit_id,
            table_name,
            operation,
            old_value,
            new_value,
            changed_by,
            change_date
        )
        VALUES(
            seq_audit_log.NEXTVAL,
            'CUSTOMER',
            'DELETE',
            'Customer ID: ' || :OLD.customer_id,
            NULL,
            USER,
            SYSDATE
        );
    END IF;

END;
