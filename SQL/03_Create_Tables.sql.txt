-- ======================================================
-- CUSTOMER TABLE
-- ======================================================

CREATE TABLE customer (
    customer_id      NUMBER PRIMARY KEY,
    first_name       VARCHAR2(50) NOT NULL,
    last_name        VARCHAR2(50) NOT NULL,
    phone            VARCHAR2(20) UNIQUE NOT NULL,
    email            VARCHAR2(100) UNIQUE,
    address          VARCHAR2(200)
);

-- ======================================================
-- VEHICLE TABLE
-- ======================================================

CREATE TABLE vehicle (
    vehicle_id       NUMBER PRIMARY KEY,
    customer_id      NUMBER NOT NULL,
    plate_number     VARCHAR2(20) UNIQUE NOT NULL,
    vehicle_type     VARCHAR2(30) NOT NULL,
    vehicle_model    VARCHAR2(50),
    CONSTRAINT fk_vehicle_customer
        FOREIGN KEY (customer_id)
        REFERENCES customer(customer_id)
);

-- ======================================================
-- EMPLOYEE TABLE
-- ======================================================

CREATE TABLE employee (
    employee_id      NUMBER PRIMARY KEY,
    first_name       VARCHAR2(50) NOT NULL,
    last_name        VARCHAR2(50) NOT NULL,
    position         VARCHAR2(40) NOT NULL,
    phone            VARCHAR2(20) UNIQUE
);

-- ======================================================
-- SERVICE TABLE
-- ======================================================

CREATE TABLE service (
    service_id       NUMBER PRIMARY KEY,
    service_name     VARCHAR2(100) NOT NULL,
    price            NUMBER(10,2) NOT NULL,
    CONSTRAINT ck_service_price
        CHECK (price > 0)
);

-- ======================================================
-- SERVICE ORDER TABLE
-- ======================================================

CREATE TABLE service_order (
    order_id         NUMBER PRIMARY KEY,
    customer_id      NUMBER NOT NULL,
    vehicle_id       NUMBER NOT NULL,
    employee_id      NUMBER NOT NULL,
    order_date       DATE DEFAULT SYSDATE,
    status           VARCHAR2(20) DEFAULT 'PENDING',

    CONSTRAINT fk_order_customer
        FOREIGN KEY (customer_id)
        REFERENCES customer(customer_id),

    CONSTRAINT fk_order_vehicle
        FOREIGN KEY (vehicle_id)
        REFERENCES vehicle(vehicle_id),

    CONSTRAINT fk_order_employee
        FOREIGN KEY (employee_id)
        REFERENCES employee(employee_id),

    CONSTRAINT ck_order_status
        CHECK (status IN ('PENDING','IN_PROGRESS','COMPLETED','PAID'))
);

-- ======================================================
-- ORDER DETAIL TABLE
-- ======================================================

CREATE TABLE order_detail (
    detail_id        NUMBER PRIMARY KEY,
    order_id         NUMBER NOT NULL,
    service_id       NUMBER NOT NULL,
    quantity         NUMBER DEFAULT 1,
    unit_price       NUMBER(10,2) NOT NULL,

    CONSTRAINT fk_detail_order
        FOREIGN KEY (order_id)
        REFERENCES service_order(order_id),

    CONSTRAINT fk_detail_service
        FOREIGN KEY (service_id)
        REFERENCES service(service_id),

    CONSTRAINT ck_quantity
        CHECK (quantity > 0),

    CONSTRAINT ck_unit_price
        CHECK (unit_price > 0)
);

-- ======================================================
-- PAYMENT TABLE
-- ======================================================

CREATE TABLE payment (
    payment_id       NUMBER PRIMARY KEY,
    order_id         NUMBER UNIQUE NOT NULL,
    payment_date     DATE DEFAULT SYSDATE,
    amount           NUMBER(10,2) NOT NULL,
    payment_method   VARCHAR2(30),

    CONSTRAINT fk_payment_order
        FOREIGN KEY (order_id)
        REFERENCES service_order(order_id),

    CONSTRAINT ck_payment_amount
        CHECK (amount > 0)
);

-- ======================================================
-- PUBLIC HOLIDAY TABLE
-- ======================================================

CREATE TABLE public_holiday (
    holiday_id       NUMBER PRIMARY KEY,
    holiday_name     VARCHAR2(100) NOT NULL,
    holiday_date     DATE UNIQUE NOT NULL
);

-- ======================================================
-- AUDIT LOG TABLE
-- ======================================================

CREATE TABLE audit_log (
    audit_id         NUMBER PRIMARY KEY,
    table_name       VARCHAR2(50),
    operation        VARCHAR2(20),
    old_value        VARCHAR2(4000),
    new_value        VARCHAR2(4000),
    changed_by       VARCHAR2(100),
    change_date      DATE DEFAULT SYSDATE
);

-- ======================================================
-- VERIFY TABLES
-- ======================================================

SELECT table_name
FROM user_tables
ORDER BY table_name;
